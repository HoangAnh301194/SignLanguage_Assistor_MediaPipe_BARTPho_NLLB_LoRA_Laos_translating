\section{Mô-đun nhận dạng ký hiệu}

% --- SLIDE 8 ---
\begin{frame}{Mô-đun 1: Tổng quan nhận dạng ký hiệu}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \begin{itemize}
            \item \textbf{Bài toán}: phân loại đa lớp trên chuỗi đặc trưng thời gian
            \item Video $\rightarrow$ chuỗi landmarks $\rightarrow$ nhãn ký hiệu (gloss)
            \item Ký hiệu phụ thuộc: vị trí tay, quỹ đạo chuyển động, biểu cảm khuôn mặt, thứ tự thời gian
            \item Không thể nhìn 1 frame đơn lẻ $\rightarrow$ cần \textbf{chuỗi multi-frame}
        \end{itemize}

        \column{0.42\textwidth}
        \begin{block}{Mô hình sử dụng}
            \small\textbf{TFLite pre-trained}\\
            Kaggle ASL Signs 1st Place\\[4pt]
            250 từ vựng ký hiệu ASL\\
            Kích thước: $\sim$11 MB\\
            Chạy trên CPU
        \end{block}
        \vspace{0.15cm}
        \begin{alertblock}{\small Lưu ý}
            \scriptsize Mô hình pre-trained, không do nhóm tự huấn luyện. Tập trung nguồn lực cho NLP phía sau.
        \end{alertblock}
    \end{columns}
\end{frame}

% --- SLIDE 9 ---
\begin{frame}{Mô-đun 1: MediaPipe Holistic -- 543 landmarks}
    \begin{columns}[T]
        \column{0.50\textwidth}
        \begin{itemize}
            \item MediaPipe Holistic (Google) trích xuất \textbf{543 landmarks} realtime
            \item Gồm: 468 khuôn mặt + 33 pose + 21 tay trái + 21 tay phải
            \item Mỗi landmark: 3 tọa độ $(x, y, z)$
            \item Tổng: \textbf{1.629 đặc trưng/frame}
        \end{itemize}

        \vspace{0.2cm}
        \small\textbf{Cấu hình MediaPipe:}
        \begin{itemize}\small
            \item \texttt{model\_complexity = 1}
            \item \texttt{min\_detection\_confidence = 0.5}
            \item \texttt{min\_tracking\_confidence = 0.5}
        \end{itemize}

        \column{0.46\textwidth}
        \centering
        \begin{tabular}{>{\small}l >{\small}r >{\small}r}
            \toprule
            \textbf{Thành phần} & \textbf{Điểm} & \textbf{Features} \\
            \midrule
            Khuôn mặt & 468 & 1.404 \\
            Tay trái & 21 & 63 \\
            Pose & 33 & 99 \\
            Tay phải & 21 & 63 \\
            \midrule
            \textbf{Tổng} & \textbf{543} & \textbf{1.629} \\
            \bottomrule
        \end{tabular}

        \vspace{0.3cm}
        \begin{alertblock}{\small Tại sao giữ cả face?}
            \scriptsize Mô hình TFLite được huấn luyện trên toàn bộ 543 landmarks. Face cung cấp tín hiệu biểu cảm quan trọng.
        \end{alertblock}
    \end{columns}
\end{frame}

        \begin{itemize}
            \item Đầu ra tầng nhận diện chỉ là gloss rời rạc:

            \item Cần biến thành câu tiếng Việt tự nhiên:
            {\small\color{deepblue} \textit{Tôi muốn uống nước.}}
            \item Bài toán: \textbf{Sequence-to-Sequence}\\
            (gloss $\rightarrow$ câu tiếng Việt)
        \end{itemize}

        \column{0.42\textwidth}
        \begin{exampleblock}{Ví dụ chuyển đổi}

            \texttt{toi muon uong\_nuoc}\\
            $\downarrow$\\

            \texttt{toi can giup\_do}\\
            $\downarrow$\\
            \textit{Tôi cần giúp đỡ.}\\[6pt]
            \texttt{toi dau bung}\\
            $\downarrow$\\
            \textit{Tôi đau bụng.}
        \end{exampleblock}
    \end{columns}
\end{frame}

\section{Tích hợp và demo}

% --- SLIDE 19 ---
\begin{frame}{Pipeline tích hợp -- realtime\_integrated.py}
    \begin{block}{Vòng lặp chính}
        \begin{enumerate}\small
            \item Đọc frame từ webcam $\rightarrow$ mô-đun nhận diện ASL-250
            \item Hiển thị top-k dự đoán để người vận hành quan sát
            \item \texttt{a}: lấy top-1, ánh xạ sang token tiếng Việt, thêm vào buffer
            \item \texttt{b}: gửi buffer $\rightarrow$ SentenceBuilder $\rightarrow$ câu Việt $\rightarrow$ dịch Lào
            \item \texttt{c}: xóa buffer, bắt đầu câu mới
            \item \texttt{SPACE}: reset bộ đệm frame nhận diện
            \item \texttt{q}: thoát, giải phóng camera
        \end{enumerate}
    \end{block}

    \vspace{0.2cm}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \begin{exampleblock}{\small Thiết kế demo}
            \scriptsize Có người vận hành hỗ trợ xác nhận token. Giảm rủi ro nhận diện nhảy nhãn. Phù hợp demo học thuật.
        \end{exampleblock}

        \column{0.48\textwidth}
        \begin{alertblock}{\small Tùy chọn --no-translation}
            \scriptsize Kiểm thử 2 tầng đầu khi máy hạn chế tài nguyên hoặc chưa tải mô hình NLLB.
        \end{alertblock}
    \end{columns}
\end{frame}
